// ========================================== // SCHOOL MANAGEMENT SYSTEM (NIDAAMKA MAAMULKA ISKUULKA) // Fasalada: Grade 1 ilaa Form 4 // ========================================== class SchoolManagementSystem { constructor() { this.students = {}; // Kaydka guud ee ardayda this.attendance = {}; // Kaydka xaadirinta (Taariikh walba iyo fasalkeeda) this.validClasses = this.generateClasses(); } // 1. Sameynta liiska fasalada (Grade 1-8 iyo Form 1-4) generateClasses() { let classes = []; for (let i = 1; i <= 8; i++) classes.push(`Grade ${i}`); for (let i = 1; i <= 4; i++) classes.push(`Form ${i}`); return classes; } // 2. MAAMULKA ARDAYDA (Xog ku daris iyo ka saaris) // Ku dar arday cusub addStudent(id, name, className) { if (!this.validClasses.includes(className)) { console.log(`❌ Cilad: Fasalka '${className}' kama mid aha fasalada iskuulka (Grade 1 - Form 4).`); return; } if (this.students[id]) { console.log(`❌ Cilad: Arday wata ID-gaan ${id} mar hore ayaa la diwangeliyey.`); return; } this.students[id] = { id: id, name: name, class: className, feePaid: 0, // Lacagta uu bixiyey feeRequired: 100 // Lacagta laga rabo (Tusaale: $100) }; console.log(`✅ Arday cusub waa la diwangeliyey: ${name} (${className})`); } // Ka saar arday iskuulka removeStudent(id) { if (!this.students[id]) { console.log(`❌ Cilad: Lama helin arday wata ID: ${id}`); return; } const studentName = this.students[id].name; delete this.students[id]; console.log(`🗑️ Ardayga ${studentName} (ID: ${id}) waa laga saaray nidaamka.`); } // 3. DAAQADDA XAADIRINTA (Attendance Management) // Xaadirinta fasal dhan taariikh gaar ah takeAttendance(date, className, attendanceRecords) { // attendanceRecords waa shay (object) leh { studentId: "Present" ama "Absent" } if (!this.validClasses.includes(className)) { console.log(`❌ Cilad: Fasalka sax ma aha.`); return; } if (!this.attendance[date]) { this.attendance[date] = {}; } this.attendance[date][className] = attendanceRecords; console.log(`📝 Xaadirintii fasalka ${className} ee taariikhda ${date} waa la kaydiyey.`); } // Soo bandhig xogta xaadirinta ee fasal gaar ah viewAttendance(date, className) { console.log(`\n--- Xogta Xaadirinta: ${className} | Taariikhda: ${date} ---`); if (!this.attendance[date] || !this.attendance[date][className]) { console.log("ℹ️ Ma jirto xog xaadirin oo laga hayo fasalkaan taariikhdaas."); return; } const records = this.attendance[date][className]; for (const [id, status] of Object.entries(records)) { const student = this.students[id]; const name = student ? student.name : "Arday la tirtiray"; console.log(`ID: ${id} | Magaca: ${name} -> Heerka: ${status === "Present" ? "✅ Jooga" : "❌ Ma Joogo"}`); } } // 4. DAAQADDA BIXINTA LACAGTA (Fee Management) // Bixinta lacagta iskuulka paySchoolFee(id, amount) { const student = this.students[id]; if (!student) { console.log(`❌ Cilad: Lama helin arday wata ID: ${id}`); return; } student.feePaid += amount; console.log(`💰 Lacag bixin: ${student.name} wuxuu bixiyey $${amount}.`); this.checkFeeStatus(id); } // Hubi heerka lacag bixinta ardayga checkFeeStatus(id) { const student = this.students[id]; if (!student) return; const balance = student.feeRequired - student.feePaid; console.log(`--- Cadaynta Lacagta: ${student.name} (${student.class}) ---`); console.log(`Laga rabay: $${student.feeRequired}`); console.log(`Bixiyey: $${student.feePaid}`); if (balance <= 0) { console.log(`Heerka: 🟢 Waa sifeeyey (Paid Fully)`); } else { console.log(`Heerka: 🔴 Waxaa ku haraadasho ah: $${balance}`); } } } // ========================================== // TUSAALE REERKA LOO ISTEMAALAYO (TESTING) // ========================================== const sms = new SchoolManagementSystem(); console.log("--- 1. Diwangelinta Ardayda ---"); sms.addStudent("S001", "Axmed Cali", "Grade 1"); sms.addStudent("S002", "Faadumo Cumar", "Form 4"); sms.addStudent("S003", "Maxamed Xasan", "Grade 5"); console.log("\n--- 2. Daandada Xaadirinta (Attendance) ---"); // Maanta oo kale (Tusaale: 2026-08-07) sms.takeAttendance("2026-08-07", "Grade 1", { "S001": "Present" }); sms.takeAttendance("2026-08-07", "Form 4", { "S002": "Absent" }); // Eegista Xaadirinta sms.viewAttendance("2026-08-07", "Form 4"); sms.viewAttendance("2026-08-07", "Grade 1"); console.log("\n--- 3. Daandada Bixinta Lacagta (Fees) ---"); sms.paySchoolFee("S002", 50); // Faadumo waxay bixisay bar lacagtii laga rabay sms.paySchoolFee("S001", 100); // Axmed wuxuu bixiyey dhamaan console.log("\n--- 4. Ka saarista iyo Wax ka badalka ---"); sms.removeStudent("S003"); // Maxamed waa laga saaray iskuulka
